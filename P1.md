```openscad
// Parametric 3D water tank with level, inlet-flow, and outlet-flow readings.
// Open in OpenSCAD and render (F6). Units are millimeters.

$fn = 64;

tank_r = 45;
tank_h = 90;
wall = 3;
base_h = 5;
pipe_r = 6;
pipe_len = 28;

module pipe_with_valve(axis = "x", z = 0, label = "FLOW") {
	// Valve body and pipe
	rotate([axis == "x" ? 0 : 90, 0, axis == "x" ? 0 : 90]) {
		translate([0, 0, 0]) cylinder(h = pipe_len, r = pipe_r);
		translate([pipe_len / 2, 0, 0]) cylinder(h = 10, r = pipe_r + 3, center = true);
		translate([pipe_len / 2, 0, -pipe_r - 6]) cube([4, 4, 18], center = true);
	}
	// Simple valve handle
	translate([pipe_len / 2, 0, -pipe_r - 14])
		cube([20, 4, 4], center = true);
	// Label plate
	translate([pipe_len + 2, 0, 0])
		rotate([90, 0, 90])
			linear_extrude(height = 1)
				text(label, size = 5, halign = "center", valign = "center");
}

module tank_shell() {
	color("deepskyblue", 0.55)
	difference() {
		cylinder(h = tank_h, r = tank_r);
		translate([0, 0, base_h])
			cylinder(h = tank_h, r = tank_r - wall);
	}
	color("silver") cylinder(h = base_h, r = tank_r + 2);
	color("silver") translate([0, 0, tank_h]) cylinder(h = 3, r = tank_r + 2);
}

module level_gauge() {
	// Vertical graduated gauge on the front of the tank
	translate([tank_r + 1, -4, 8]) {
		color("white") cube([3, 8, tank_h - 16]);
		for (i = [0:10])
			translate([-1, 0, i * (tank_h - 16) / 10])
				color("black") cube([7, 8, 1]);
		color("red") translate([-2, 0, tank_h * 0.62]) cube([8, 8, 3]);
		translate([7, 0, tank_h * 0.62])
			rotate([90, 0, 90])
				linear_extrude(height = 1)
					text("LEVEL 62%", size = 5, halign = "left", valign = "center");
	}
}

module top_sensor() {
	color("orange") {
		translate([0, 0, tank_h + 3]) cylinder(h = 12, r = 4);
		translate([0, 0, tank_h + 15]) sphere(r = 5);
	}
	translate([7, 0, tank_h + 12])
		rotate([90, 0, 90])
			linear_extrude(height = 1)
				text("LEVEL SENSOR", size = 4, halign = "left", valign = "center");
}

// Main assembly
tank_shell();
level_gauge();
top_sensor();

// Inlet valve at the lower-left and outlet valve at the upper-right.
translate([-tank_r - pipe_len, 0, 18]) rotate([0, 0, 180])
	pipe_with_valve("x", 18, "IN  12 L/min");
translate([tank_r, 0, 65]) pipe_with_valve("x", 65, "OUT  8 L/min");

// Liquid volume indicator inside the transparent tank
color("#1687d9", 0.8)
translate([0, 0, base_h]) cylinder(h = tank_h * 0.62, r = tank_r - wall - 1);
```

## Design features

- Cylindrical transparent tank with a removable top and reinforced base.
- Graduated front gauge showing the tank level; the example reading is **62%**.
- Orange top-mounted level sensor.
- Lower inlet valve labeled **IN 12 L/min**.
- Upper outlet valve labeled **OUT 8 L/min**.
- Valve handles and raised labels are included for 3D-printing or visualization.

The model is written in OpenSCAD. Change `tank_h`, `tank_r`, the liquid height, or the flow labels to represent different readings.
